        const vsSource = `
            attribute vec3 aPos;
            attribute vec2 aTexCoord;
            attribute vec3 aNormal;
            attribute float aAO;
            
            uniform mat4 uView;
            uniform mat4 uProj;
            
            varying vec2 vTexCoord;
            varying float vLight;
            varying float vAO;
            varying float vFog;
            
            void main() {
                vec4 worldPos = vec4(aPos, 1.0);
                vec4 viewPos = uView * worldPos;
                gl_Position = uProj * viewPos;
                
                vTexCoord = aTexCoord;
                vec3 lightDir = normalize(vec3(0.5, 1.0, 0.3));
                vLight = max(dot(aNormal, lightDir), 0.3);
                vAO = aAO;
                
                float dist = length(viewPos.xyz);
                vFog = clamp((dist - 50.0) / 50.0, 0.0, 1.0);
            }
        `;

        const fsSource = `
            precision mediump float;
            
            varying vec2 vTexCoord;
            varying float vLight;

            uniform float uTime;

            void main() {
                vec2 tc = fract(vTexCoord * 4.0);
                float pattern = step(0.05, tc.x) * step(0.05, tc.y) * 
                               step(tc.x, 0.95) * step(tc.y, 0.95);
                
                vec3 baseColor = uColor * (0.8 + pattern * 0.2);
                vec3 litColor = baseColor * vLight * vAO;
